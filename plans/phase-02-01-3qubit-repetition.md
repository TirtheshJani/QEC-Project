**Goal:** Build the 3-qubit bit-flip repetition code with density-matrix simulation under independent single-qubit depolarizing noise, produce a logical-error vs physical-error plot, and promote a reusable `ThreeQubitBitFlipCode` helper class for downstream concatenation (Phase 2.2 Shor).

**Architecture:**
1. Promote a `ThreeQubitBitFlipCode` class to `src/qec_project/codes/repetition3.py` that exposes `encode`, `apply_noise`, `syndrome_extract`, and `decode`. Phase 2.2 will reuse the encoder when concatenating with a phase-flip code into Shor 9.
2. Promote single-qubit Kraus operators (`bit_flip_kraus`, `depolarizing_channel`) and a generic `apply_channel` helper to `src/qec_project/noise/quantum.py`. Phase 2 needs them; downstream phases will too.
3. Drive the pedagogy from a buildable notebook (`three_qubit_repetition.ipynb`) generated by `_build_notebook.py` so diffs are reviewable.

**Tech Stack:** numpy (matrices, RNG), pytest + hypothesis (tests), matplotlib (plot), nbformat (notebook builder). No new dependencies beyond the existing `pyproject.toml` `dev` extra.

**Decoding protocol (locked):** We simulate one round per p sample using a *logical-basis ensemble*: for each of N=2000 trials we sample one of the four basis logical states |0_L>, |1_L>, |+_L>, |-_L> uniformly, encode it as a pure state, apply independent single-qubit depolarizing channels to each physical qubit, extract a stabilizer syndrome (sampled from the diagonal of the noisy density matrix in the Z basis), apply the maximum-likelihood single-qubit X recovery, decode back to the logical qubit, and compare overlap with the input. The uncoded baseline applies the same depolarizing channel to a single qubit carrying the same logical state. The metric is the empirical infidelity averaged across the four basis states (a 2-design average, which equals the average channel infidelity).

**Why density-matrix + sampled-syndrome rather than full Stinespring/ancilla circuits:** Phase 2.2 (Shor 9-qubit, the next agent's work) introduces ancilla-based syndrome extraction using Qiskit. Here we want the conceptual picture (stabilizer eigenspaces project the state) without the circuit machinery.

```python
"""Single-qubit Pauli matrices and Kraus channels.

Used directly by the 3-qubit repetition notebook (Phase 2.1) and downstream
by Shor 9 (Phase 2.2). Kept tiny and dependency-free so unit tests can
property-check trace preservation and Kraus completeness cheaply.
"""

from __future__ import annotations

from collections.abc import Sequence

import numpy as np
from numpy.typing import NDArray

PAULI_I: NDArray[np.complex128] = np.eye(2, dtype=complex)
PAULI_X: NDArray[np.complex128] = np.array([[0, 1], [1, 0]], dtype=complex)
PAULI_Y: NDArray[np.complex128] = np.array([[0, -1j], [1j, 0]], dtype=complex)
PAULI_Z: NDArray[np.complex128] = np.array([[1, 0], [0, -1]], dtype=complex)


def bit_flip_kraus(p: float) -> list[NDArray[np.complex128]]:
    """Single-qubit bit-flip channel: rho -> (1-p) rho + p X rho X.

    Returns Kraus operators [K0, K1] with K0 = sqrt(1-p) I, K1 = sqrt(p) X.
    """
    if not 0.0 <= p <= 1.0:
        raise ValueError(f"p must be in [0, 1]; got {p}")
    return [np.sqrt(1.0 - p) * PAULI_I, np.sqrt(p) * PAULI_X]


def depolarizing_channel(p: float) -> list[NDArray[np.complex128]]:
    """Single-qubit depolarizing channel.

    rho -> (1 - p) rho + (p/3)(X rho X + Y rho Y + Z rho Z)

    At p = 1 this maps every state to the maximally mixed state I/2.
    Kraus operators: [sqrt(1-p) I, sqrt(p/3) X, sqrt(p/3) Y, sqrt(p/3) Z].
    """
    if not 0.0 <= p <= 1.0:
        raise ValueError(f"p must be in [0, 1]; got {p}")
    return [
        np.sqrt(1.0 - p) * PAULI_I,
        np.sqrt(p / 3.0) * PAULI_X,
        np.sqrt(p / 3.0) * PAULI_Y,
        np.sqrt(p / 3.0) * PAULI_Z,
    ]


def apply_channel(
    rho: NDArray[np.complex128],
    kraus: Sequence[NDArray[np.complex128]],
) -> NDArray[np.complex128]:
    """Apply a Kraus channel to a density matrix: rho -> sum_k K_k rho K_k^dagger."""
    out = np.zeros_like(rho)
    for K in kraus:
        out = out + K @ rho @ K.conj().T
    return out


def apply_channel_to_qubit(
    rho: NDArray[np.complex128],
    kraus: Sequence[NDArray[np.complex128]],
    target: int,
    n_qubits: int,
) -> NDArray[np.complex128]:
    """Apply a single-qubit Kraus channel to qubit `target` of an n-qubit state.

    Qubit ordering: qubit 0 is the leftmost (most significant) tensor factor,
    matching numpy.kron(a, b) where a acts on qubit 0.
    """
    if not 0 <= target < n_qubits:
        raise ValueError(f"target {target} not in [0, {n_qubits})")
    dim = 2**n_qubits
    if rho.shape != (dim, dim):
        raise ValueError(f"rho shape {rho.shape} does not match {n_qubits} qubits")

    left = np.eye(2**target, dtype=complex)
    right = np.eye(2 ** (n_qubits - target - 1), dtype=complex)
    lifted = [np.kron(np.kron(left, K), right) for K in kraus]
    return apply_channel(rho, lifted)
```

```python
"""The 3-qubit bit-flip repetition code.

Encoding:
    a |0> + b |1>  ->  a |000> + b |111>

Stabilizers (computational basis, qubit 0 leftmost):
    S0 = Z (x) Z (x) I
    S1 = I (x) Z (x) Z

Logical operators:
    X_L = X (x) X (x) X
    Z_L = Z (x) Z (x) Z  (any one Z_i suffices on the code space; we use the symmetric form)

Single-qubit X errors are detectable and uniquely identified by the
two-bit syndrome (s0, s1):
    (0, 0) -> no error
    (1, 0) -> X on qubit 0
    (1, 1) -> X on qubit 1
    (0, 1) -> X on qubit 2

Z and Y errors are NOT corrected (they commute or anticommute with the
stabilizers in a way that does not pin them down): a single Z is undetectable
because both stabilizers are products of Z's. This is the headline limitation
that motivates Shor 9 in Phase 2.2.
"""

from __future__ import annotations

from collections.abc import Sequence

import numpy as np
from numpy.typing import NDArray

from qec_project.noise.quantum import (
    PAULI_I,
    PAULI_X,
    PAULI_Z,
    apply_channel_to_qubit,
    depolarizing_channel,
)

# Recovery table: (s0, s1) -> qubit index to flip with X (or -1 for "no flip").
_SYNDROME_RECOVERY: dict[tuple[int, int], int] = {
    (0, 0): -1,
    (1, 0): 0,
    (1, 1): 1,
    (0, 1): 2,
}


class ThreeQubitBitFlipCode:
    """3-qubit bit-flip repetition code with density-matrix simulation."""

    n_qubits: int = 3
    k_qubits: int = 1
    distance: int = 3

    def encode(self, state_1q: NDArray[np.complex128]) -> NDArray[np.complex128]:
        """Map a single-qubit state vector (length 2) to the 3-qubit code space.

        |0> -> |000>, |1> -> |111>, linear in between.
        """
        if state_1q.shape != (2,):
            raise ValueError(f"state_1q must have shape (2,); got {state_1q.shape}")
        out = np.zeros(8, dtype=complex)
        out[0] = state_1q[0]  # |000>
        out[7] = state_1q[1]  # |111>
        return out
```

```python
    @staticmethod
    def stabilizers() -> tuple[NDArray[np.complex128], NDArray[np.complex128]]:
        """Return the two stabilizer generators as 8x8 matrices.

        S0 = Z (x) Z (x) I  detects X on qubit 0 or qubit 1.
        S1 = I (x) Z (x) Z  detects X on qubit 1 or qubit 2.
        """
        S0 = np.kron(np.kron(PAULI_Z, PAULI_Z), PAULI_I)
        S1 = np.kron(np.kron(PAULI_I, PAULI_Z), PAULI_Z)
        return S0, S1

    @staticmethod
    def logical_x() -> NDArray[np.complex128]:
        """X_L = X (x) X (x) X."""
        return np.kron(np.kron(PAULI_X, PAULI_X), PAULI_X)

    @staticmethod
    def logical_z() -> NDArray[np.complex128]:
        """Z_L = Z (x) Z (x) Z (any single Z_i acts identically on the code space)."""
        return np.kron(np.kron(PAULI_Z, PAULI_Z), PAULI_Z)

    def syndrome_extract(
        self,
        rho: NDArray[np.complex128],
        rng: np.random.Generator,
    ) -> tuple[int, int]:
        """Project rho onto a joint eigenspace of (S0, S1) and return the
        outcome (s0, s1) in {0, 1} x {0, 1}.

        The four eigenspaces partition the 8-dim Hilbert space:
            (0, 0): {|000>, |111>}     -- code space
            (1, 0): {|100>, |011>}     -- X error on qubit 0
            (1, 1): {|010>, |101>}     -- X error on qubit 1
            (0, 1): {|001>, |110>}     -- X error on qubit 2

        Outcome probabilities are read off the diagonal of rho in the
        computational basis; rng selects an outcome stochastically.

        NOTE (pedagogy): on real hardware, syndromes are measured
        non-destructively via ancilla qubits and CNOT chains. Phase 2.2
        builds that ancilla circuit explicitly in Qiskit. Here we shortcut
        through the density-matrix picture to keep the focus on what the
        syndrome *means*.
        """
        diag = np.real(np.diag(rho))
        diag = np.clip(diag, 0.0, None)
        total = diag.sum()
        if total <= 0:
            raise ValueError("density matrix has non-positive trace")
        probs = diag / total

        # Map computational basis index -> (s0, s1).
        # index bits b0 b1 b2 (b0 leftmost). Z on qubit i has eigenvalue (-1)**bi.
        # s_i = 1 when the corresponding ZZ product = -1 = bits XOR.
        idx = int(rng.choice(8, p=probs))
        b = [(idx >> 2) & 1, (idx >> 1) & 1, idx & 1]
        s0 = b[0] ^ b[1]
        s1 = b[1] ^ b[2]
        return s0, s1
```

```python
    def apply_noise(
        self,
        rho: NDArray[np.complex128],
        p: float,
        rng: np.random.Generator,  # noqa: ARG002 -- accepted for signature symmetry; channel is deterministic
    ) -> NDArray[np.complex128]:
        """Apply independent single-qubit depolarizing noise with parameter p
        to each of the 3 physical qubits.

        The channel is deterministic in the density-matrix picture; `rng` is
        accepted only to match the `apply_noise(rho, p, rng)` signature used
        across the codebase (some channels will be stochastic in Phase 5).
        """
        K = depolarizing_channel(p)
        out = rho
        for q in range(self.n_qubits):
            out = apply_channel_to_qubit(out, K, target=q, n_qubits=self.n_qubits)
        return out

    def decode(
        self,
        rho: NDArray[np.complex128],
        syndrome: tuple[int, int],
    ) -> NDArray[np.complex128]:
        """Apply the maximum-likelihood single-qubit X recovery for the
        observed syndrome and return the corrected density matrix.
        """
        q = _SYNDROME_RECOVERY.get(tuple(syndrome))
        if q is None:
            raise ValueError(f"unknown syndrome {syndrome}")
        if q == -1:
            return rho
        # Build the lifted X_q operator on 3 qubits.
        ops: list[NDArray[np.complex128]] = [PAULI_I, PAULI_I, PAULI_I]
        ops[q] = PAULI_X
        X_q = np.kron(np.kron(ops[0], ops[1]), ops[2])
        return X_q @ rho @ X_q.conj().T
```

```python
    # Four basis logical states used to estimate the average channel infidelity.
    _LOGICAL_BASIS: tuple[tuple[str, NDArray[np.complex128]], ...] = (
        ("0", np.array([1.0, 0.0], dtype=complex)),
        ("1", np.array([0.0, 1.0], dtype=complex)),
        ("+", np.array([1.0, 1.0], dtype=complex) / np.sqrt(2.0)),
        ("-", np.array([1.0, -1.0], dtype=complex) / np.sqrt(2.0)),
    )

    def _decode_to_logical_density_matrix(
        self, rho: NDArray[np.complex128]
    ) -> NDArray[np.complex128]:
        """Project a 3-qubit density matrix onto the code space and read off
        the corresponding single-qubit (2x2) logical density matrix.

        After ML recovery the corrected state is supported on the code space
        (up to bit-flip residue from undetectable two-qubit errors). We take
        the 2x2 block in the {|000>, |111>} basis. Off-code-space weight is
        renormalized into the logical subspace.
        """
        rho_log = np.array(
            [[rho[0, 0], rho[0, 7]], [rho[7, 0], rho[7, 7]]], dtype=complex
        )
        tr = np.real(np.trace(rho_log))
        if tr <= 0:
            # All weight escaped the code space (extreme noise). Fall back to
            # maximally mixed -> infidelity 1/2 against any pure state.
            return 0.5 * np.eye(2, dtype=complex)
        return rho_log / tr

    def simulate_logical_error_rate(
        self,
        p: float,
        n_trials: int,
        rng: np.random.Generator,
    ) -> float:
        """Monte-Carlo estimate of average logical infidelity at physical
        error rate `p`.

        Protocol per trial:
            1. Sample a logical state uniformly from {|0>, |1>, |+>, |->}.
            2. Encode -> 3-qubit pure state -> density matrix.
            3. Apply independent single-qubit depolarizing(p) on each qubit.
            4. Sample a stabilizer syndrome from the Z-basis diagonal.
            5. Project the (post-syndrome) density matrix into the matching
               eigenspace and renormalize.
            6. Apply ML recovery.
            7. Read out the logical 2x2 density matrix.
            8. Trial fails if the logical fidelity <psi|rho_L|psi> < 0.5.

        The empirical failure rate is the returned logical error rate.
        """
        if n_trials <= 0:
            raise ValueError("n_trials must be positive")
        failures = 0
        for _ in range(n_trials):
            _, state_1q = self._LOGICAL_BASIS[int(rng.integers(0, 4))]
            psi = self.encode(state_1q)
            rho = np.outer(psi, psi.conj())
            rho = self.apply_noise(rho, p=p, rng=rng)
            syn = self.syndrome_extract(rho, rng)
            rho = self._project_to_syndrome_subspace(rho, syn)
            rho = self.decode(rho, syndrome=syn)
            rho_log = self._decode_to_logical_density_matrix(rho)
            fidelity = float(np.real(state_1q.conj() @ rho_log @ state_1q))
            if fidelity < 0.5:
                failures += 1
        return failures / n_trials

    @staticmethod
    def _project_to_syndrome_subspace(
        rho: NDArray[np.complex128], syndrome: tuple[int, int]
    ) -> NDArray[np.complex128]:
        """Project onto the joint eigenspace of (S0, S1) labelled by syndrome
        and renormalize. The four eigenspaces are 2-dim and are spanned by
        pairs of computational basis states (see syndrome_extract docstring).
        """
        # Which two computational basis indices belong to this syndrome.
        members: list[int] = []
        for idx in range(8):
            b0, b1, b2 = (idx >> 2) & 1, (idx >> 1) & 1, idx & 1
            s0 = b0 ^ b1
            s1 = b1 ^ b2
            if (s0, s1) == tuple(syndrome):
                members.append(idx)
        P = np.zeros((8, 8), dtype=complex)
        for m in members:
            P[m, m] = 1.0
        rho_proj = P @ rho @ P
        tr = np.real(np.trace(rho_proj))
        if tr <= 1e-15:
            # Shouldn't happen for a syndrome we just sampled; guard anyway.
            return rho
        return rho_proj / tr
```

```python
            "        np.array([1.0, 0.0], dtype=complex),\n"
            "        np.array([0.0, 1.0], dtype=complex),\n"
            "        np.array([1.0, 1.0], dtype=complex) / np.sqrt(2),\n"
            "        np.array([1.0, -1.0], dtype=complex) / np.sqrt(2),\n"
            "    ]\n"
            "    K = depolarizing_channel(p)\n"
            "    failures = 0\n"
            "    for _ in range(n_trials):\n"
            "        psi = basis[int(rng.integers(0, 4))]\n"
            "        rho = np.outer(psi, psi.conj())\n"
            "        rho_n = apply_channel(rho, K)\n"
            "        fidelity = float(np.real(psi.conj() @ rho_n @ psi))\n"
            "        if fidelity < 0.5:\n"
            "            failures += 1\n"
            "    return failures / n_trials"
        )
    )

    cells.append(
        code(
```
